\begin{figure}
\includegraphics[width=\textwidth]{test-energy.pdf}
\caption{Time, power and energy to train one C4.5 tree on the ECBDL'14 15M
subsample (depth $\le$ 10, unpruned). Each row is one implementation on one
server: the CPU implementations on epyc and dell, the GPU implementations on
the A100 and H100 servers. From left to right: training time; mean power of
the whole server while training, with the share the idle server would draw in a
lighter shade; the part of that power drawn by the CPUs and the GPU in use
(hatched: the CPUs during a GPU run); and the energy per tree, time $\times$
power. Time and energy are on logarithmic scales.}
\label{fig:energy}
\end{figure}

\begin{figure}
\includegraphics[width=\textwidth]{test-energy-mnist8m.pdf}
\caption{Time, power and energy to train one tree on MNIST8M without a depth
limit (unpruned): \ImplParallel{}, \ImplGpu{} and \ImplHybrid{} on the same H100
server, all 32 CPUs. Panels as in Fig.~\ref{fig:energy}, on linear scales: the
lighter shade in the server power and the energy is the idle server's share;
in the CPU+GPU power the hatched part is the CPU packages, the rest the GPU in
use (none for \ImplParallel{}). Means of six rounds of 5--17 trees each
(standard deviation of the energy at most 0.6\,\%).}
\label{fig:energy-deep}
\end{figure}

\begin{figure}
\includegraphics[width=\textwidth]{diagram-task-parallel.pdf}
\caption{Task parallelism in \ImplParallel{} (schematic). (a)~A small tree; box
width is proportional to the node's samples, colour shows the thread that grows
it, dark nodes are split by attribute on all threads. (b)~The four threads over
time: after splitting a node of at least 4,096 samples, a thread passes one
child to the pool (arrows) and continues with the other.}
\label{fig:task-parallel}
\end{figure}

\begin{figure}
\includegraphics[width=\textwidth]{diagram-small-nodes.pdf}
\caption{Small nodes in a deep tree (HIGGS, CART, depth $\le$ 30, unpruned).
(a)~Samples of each level by the size of their node. (b)~With $m = 512$,
\ImplHybrid{} grows 97\% of the nodes from level 20 on on the CPU, but only
50--71\% of the samples. (c)~Schematic: after the hand-offs (orange), CPU
threads grow the small subtrees while the GPU's remaining levels get shorter.}
\label{fig:small-nodes}
\end{figure}
